# ============================================================================
# Доступ к папке с фото с другого домена (для генераторов NEWS DROP)
# ----------------------------------------------------------------------------
# Положи этот файл на minskmafia.by по пути:
#     /www/minskmafia.by/content/photo/.htaccess
# Именно .htaccess, с точкой в начале и без расширения .txt
#
# Без него картинки на странице покажутся, но при экспорте в PNG вместо них
# будет серая плашка: браузер запрещает считывать пиксели чужих изображений.
#
# Это для Apache. Если хостинг на nginx, .htaccess не работает — тогда нужна
# строка add_header Access-Control-Allow-Origin *; в конфиге, её ставит
# поддержка хостинга. Спроси у них, какой сервер.
# ============================================================================

<IfModule mod_headers.c>
    <FilesMatch "\.(png|jpe?g|gif|webp|svg|avif)$">
        Header set Access-Control-Allow-Origin "*"
        Header set Cross-Origin-Resource-Policy "cross-origin"
    </FilesMatch>
</IfModule>
